
### 完整配置 (已完成的值)

```yaml
# ============================================================
# Galaxy Cloud Executor - AWS 云配置
# ============================================================

cloud_providers:
  aws:
    # ========== 基础配置 (必须) ==========
    enabled: true
    region: "us-west-2"                              # ✅ 已配置
    batch_queue: "galaxy-cloud-queue"                # ✅ 已创建
    s3_bucket: "galaxy-cloud-data-1760241377"        # ✅ 已创建

    # ========== IAM 角色 (必须!) ==========
    job_role_arn: "arn:aws:iam::399683337364:role/BatchJobRole"  # ✅ 已配置
    # 重要: 容器必须有此角色才能访问 S3

    # ========== Spot 实例配置 (可选) ==========
    use_spot: false  # 设置为 true 可节省 60-90% 成本
    spot_bid_percentage: 100  # Spot 实例出价 (On-Demand 价格的百分比)
    # 注意: Spot 实例可能被中断，适合可容错的作业

    # ========== 日志配置 ==========
    log_retention_days: 7  # CloudWatch Logs 保留天数
    # 可选值: 1, 3, 5, 7, 14, 30, 60, 90, 120, 150, 180, 365, 400, 545, 731, 1827, 3653

    # ========== 作业配置 ==========
    job_timeout_seconds: 172800  # 作业超时时间 (48小时)
    max_retries: 2               # 作业失败后重试次数

    # ========== 数据传输配置 ==========
    s3_multipart_threshold_mb: 100  # 大于此大小的文件使用 multipart upload
    s3_multipart_chunksize_mb: 50   # Multipart upload 的分块大小

    # ========== 高级配置 ==========
    # AWS 凭证配置 (可选，如果不使用环境变量或 AWS CLI 配置)
    # aws_access_key_id: ""        # 留空使用默认凭证链
    # aws_secret_access_key: ""    # 留空使用默认凭证链
    # aws_session_token: ""        # 仅在使用临时凭证时需要

# ============================================================
# 配置说明
# ============================================================
#
# 1. 必须配置项:
#    - region, batch_queue, s3_bucket, job_role_arn
#
# 2. job_role_arn 是最关键的配置!
#    - 容器需要此角色访问 S3
#    - 如果未配置，作业会失败 (403 错误)
#
# 3. AWS 凭证优先级:
#    - 配置文件中的 aws_access_key_id/aws_secret_access_key
#    - 环境变量 (AWS_ACCESS_KEY_ID, AWS_SECRET_ACCESS_KEY)
#    - AWS CLI 配置 (~/.aws/credentials)
#    - EC2 Instance Profile (如果在 EC2 上运行)
#
# 4. Spot 实例:
#    - 成本节省: 60-90%
#    - 风险: 可能被中断 (通常 <5% 概率)
#    - 推荐: 非关键作业启用
#
# ============================================================
```

```yaml
# ============================================================
# Galaxy Job Configuration with Cloud Executor
# ============================================================

# ========== Job Runners ==========
runners:
  # 本地执行
  local:
    load: galaxy.jobs.runners.local:LocalJobRunner
    workers: 4

  # 云执行 (AWS Batch)
  galaxy_cloud:
    load: galaxy_cloud.galaxy_workflow.galaxy_job_runner:GalaxyCloudJobRunner
    workers: 16  # 并发作业处理线程数
    cloud_config_file: config/aws_cloud_config.yaml
    default_provider: aws

# ========== Execution Environments ==========
execution:
  default: local  # 默认使用本地执行

  environments:
    # ========== 本地执行 ==========
    local:
      runner: local
      tmp_dir: true

    # ========== 云执行环境 ==========

    # 云执行 - 默认配置
    cloud_default:
      runner: galaxy_cloud
      provider: aws
      queue_name: galaxy-cloud-queue    # ✅ 已创建
      vcpus: 2
      memory_mb: 4096
      disk_gb: 50
      embed_metadata_in_job: false

    # 云执行 - 小型任务
    # 适合: FastQC, SAMtools, Cutadapt
    cloud_small:
      runner: galaxy_cloud
      provider: aws
      queue_name: galaxy-cloud-queue
      vcpus: 1
      memory_mb: 2048
      disk_gb: 20

      runner: galaxy_cloud
```

## 工具映射策略

### 策略 1: 保守策略 (推荐)

只配置确定要云执行的工具:

```yaml
tools:
  # 明确列出每个工具
  - id: toolshed.g2.bx.psu.edu/repos/devteam/fastqc/fastqc/0.73+galaxy0
    environment: cloud_small

  - id: toolshed.g2.bx.psu.edu/repos/devteam/bwa/bwa_mem/0.7.17.2
    environment: cloud_medium

  # 其他工具默认本地执行
  - class: default
    environment: local
```

### 策略 2: 激进策略

默认所有工具云执行，仅保留特定工具本地:

```yaml
tools:
  # 系统工具本地执行
  - class: local
    environment: local

  - id: upload1
    environment: local

  # 其他所有工具默认云执行
  - class: default
    environment: cloud_default
```

### 策略 3: 混合策略

根据工具类型自动分配:

```yaml
tools:
  # 按工具仓库批量配置
  - id: toolshed.g2.bx.psu.edu/repos/devteam/fastqc/*
    environment: cloud_small

  - id: toolshed.g2.bx.psu.edu/repos/devteam/bwa/*
    environment: cloud_medium

  - id: toolshed.g2.bx.psu.edu/repos/iuc/gatk4_*
    environment: cloud_large

  # 保留本地执行
  - class: local
    environment: local
```
