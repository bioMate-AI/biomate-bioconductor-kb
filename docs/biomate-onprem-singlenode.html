1. 适用场景与架构概述

适用场景

本手册适用于以下部署场景：

  • 客户环境为全封闭内网，服务器无任何出网通道
  • 单台高性能服务器（推荐 32 核以上 / 128 GB 以上内存）
  • 客户已有或自行部署了本地大模型服务（vLLM / LMDeploy 等，OpenAI 兼容接口）
  • 由 BioMate 团队上门部署，客户不需要了解技术细节
ℹ️
单机 vs 多机 本手册仅覆盖单机方案。多机集群（k3s）或 HPC（Slurm）请参考对应手册。 单机方案中，所有 Nextflow 工作流均在本机 Docker 容器中运行，无需任何外部调度器。

3. 部署前准备（在联网机器上操作）

以下步骤在有网络访问权限的机器（非目标服务器）上执行，准备好所有离线材料。

3.1 下载 Docker 镜像

在联网机器上拉取并导出所有需要的镜像：

# 创建工作目录
mkdir -p ~/biomate-offline-package/images
cd ~/biomate-offline-package

# ── 1. BioMate 服务镜像 ────────────────────────────────────
# 从 BioMate 团队获取具体的镜像 tag（如 v2.5.0）
GALAXY_TAG=v2.x.x    # 替换为实际版本
FRONTEND_TAG=v2.x.x  # 替换为实际版本

docker pull 399683337364.dkr.ecr.us-west-2.amazonaws.com/biomate-galaxy:${GALAXY_TAG}
docker pull 399683337364.dkr.ecr.us-west-2.amazonaws.com/biomate-frontend:${FRONTEND_TAG}

docker save 399683337364.dkr.ecr.us-west-2.amazonaws.com/biomate-galaxy:${GALAXY_TAG} \
  | gzip > images/biomate-galaxy.tar.gz

docker save 399683337364.dkr.ecr.us-west-2.amazonaws.com/biomate-frontend:${FRONTEND_TAG} \
  | gzip > images/biomate-frontend.tar.gz

# ── 2. 基础服务镜像 ────────────────────────────────────────
for image in \
  "postgres:15" \
  "redis:7-alpine" \
  "minio/minio:latest" \
  "minio/mc:latest" \
  "registry:2"; do
  name=$(echo $image | tr '/:' '--')
  docker pull $image
  docker save $image | gzip > images/${name}.tar.gz
  echo "已保存: ${name}.tar.gz"
done

# ── 3. 生信工具镜像（按需选取）─────────────────────────────
# 查看 BioMate 工作流目录获取完整列表
# 常用生信工具镜像（示例）：
for image in \
  "biocontainers/bwa:v0.7.17" \
  "biocontainers/star:2.7.10a" \
  "bioconductor/bioconductor_docker:RELEASE_3_18" \
  "nfcore/base:2.1" \
  "quay.io/biocontainers/samtools:1.18"; do
  name=$(echo $image | tr '/:' '--')
  docker pull $image
  docker save $image | gzip > images/${name}.tar.gz
  echo "已保存: ${name}.tar.gz"
done

3.2 打包完整部署包

# 克隆 BioMate 代码（或使用 BioMate 提供的 release 压缩包）
git clone https://github.com/biomate/biomate.git biomate-src
# 或解压 BioMate 提供的 biomate-v2.x.x.tar.gz

# 下载 Nextflow（离线版）
mkdir -p ~/biomate-offline-package/tools
cd ~/biomate-offline-package/tools

# 下载 Nextflow 二进制（含所有 JAR 依赖）
curl -fsSL https://github.com/nextflow-io/nextflow/releases/download/v24.10.5/nextflow \
  -o nextflow
chmod +x nextflow

# 下载 Java 17（离线包）
curl -fsSL https://download.java.net/java/GA/jdk17/jdk-17_linux-x64_bin.tar.gz \
  -o jdk17.tar.gz

# ── 打包成单一传输包 ────────────────────────────────────────
cd ~
tar -czf biomate-offline-$(date +%Y%m%d).tar.gz \
  biomate-offline-package/ \
  biomate-src/

ls -lh biomate-offline-*.tar.gz
# 预期大小：约 15~50 GB（取决于生信工具镜像数量）
✅
部署包清单 确保以下文件都在包中：
images/biomate-galaxy.tar.gz   images/biomate-frontend.tar.gz   images/postgres--15.tar.gz   images/redis--7-alpine.tar.gz   images/minio-minio--latest.tar.gz   images/registry--2.tar.gz   tools/nextflow   tools/jdk17.tar.gz   biomate-src/docker-compose.onprem.yml   biomate-src/install.sh

4. 传输部署包到目标服务器

将打包文件通过安全方式传输到目标服务器（U盘、内网共享、SCP 等）。

# 方式 A：通过内网 SCP 传输
scp biomate-offline-20260916.tar.gz user@目标服务器IP:/opt/

# 方式 B：U 盘 / 移动硬盘（推荐用于完全隔离的内网）
# 直接拷贝到 /opt/ 目录

# 方式 C：内网文件共享服务器
# 在目标服务器上执行：
curl -O http://内网文件服务器IP/biomate-offline-20260916.tar.gz
⚠️
校验文件完整性 传输完成后建议校验 MD5/SHA256，避免因传输中断导致安装失败：
sha256sum biomate-offline-20260916.tar.gz 与联网机器上生成的 SHA256 对比。

5. 目标服务器安装步骤

以下所有操作在目标服务器上以 root 或具备 sudo 权限的用户执行。

步骤 1：解压部署包

root@server# mkdir -p /opt/biomate && cd /opt/biomate
root@server# tar -xzf /opt/biomate-offline-20260916.tar.gz --strip-components=1
root@server# ls
biomate-src/   images/   tools/

步骤 2：安装 Docker（离线）

如果目标服务器已安装 Docker ≥ 24.0，跳过此步骤。

# Ubuntu 22.04 为例（其他系统参考 Docker 官方离线安装文档）

# 检查现有 Docker 版本
docker version 2>/dev/null || echo "未安装"

# 如需安装，使用离线 deb 包（需在联网机器上提前下载）
# 下载命令（在联网机器执行）：
# apt-get download docker-ce docker-ce-cli containerd.io docker-compose-plugin
# 然后打包随部署包一起传入

# 安装（在目标服务器执行）
dpkg -i /opt/biomate/tools/docker-packages/*.deb
systemctl enable --now docker
docker version  # 验证

步骤 3：安装 Java 17（Nextflow 依赖）

# 解压 JDK 到 /usr/local/java
mkdir -p /usr/local/java
tar -xzf /opt/biomate/tools/jdk17.tar.gz -C /usr/local/java --strip-components=1

# 配置环境变量
cat >> /etc/environment << 'EOF'
JAVA_HOME=/usr/local/java
PATH=/usr/local/java/bin:$PATH
EOF

export JAVA_HOME=/usr/local/java
export PATH=$JAVA_HOME/bin:$PATH

java -version  # 应显示 openjdk 17

步骤 4：安装 Nextflow

# 安装到系统路径
cp /opt/biomate/tools/nextflow /usr/local/bin/nextflow
chmod +x /usr/local/bin/nextflow

# 禁用 Nextflow 自动更新（内网环境）
mkdir -p ~/.nextflow
cat > ~/.nextflow/config << 'EOF'
nextflow.enable.dsl = 2
EOF

# 设置离线模式环境变量（写入 /etc/environment）
echo 'NXF_OFFLINE=true' >> /etc/environment
echo 'NXF_HOME=/opt/biomate/nextflow-home' >> /etc/environment

nextflow -version  # 验证

步骤 5：加载 Docker 镜像

root@server# cd /opt/biomate/images

# 批量加载所有镜像
for f in *.tar.gz; do
  echo "加载: $f"
  docker load -i "$f"
done

# 验证镜像已加载
docker images | grep -E "biomate|postgres|redis|minio|registry"
# 重新打 tag（便于 docker-compose 引用）
docker tag 399683337364.dkr.ecr.us-west-2.amazonaws.com/biomate-galaxy:v2.x.x biomate-galaxy:latest
docker tag 399683337364.dkr.ecr.us-west-2.amazonaws.com/biomate-frontend:v2.x.x biomate-frontend:latest

步骤 6：配置 .env 文件

root@server# cd /opt/biomate/biomate-src
root@server# cp .env.onprem.example .env
root@server# nano .env  # 或 vi .env

必须填写的关键配置项：

# ─── 1. 许可证（BioMate 团队提供）─────────────────────────────
BIOMATE_LICENSE_KEY=BMOP-XXXX-XXXX-XXXX-XXXX
BIOMATE_LICENSE_SECRET=（BioMate提供）

# ─── 2. 数据库密码（自定义，至少16位）────────────────────────
POSTGRES_PASSWORD=MyStr0ng#DB#Pass2026

# ─── 3. MinIO 对象存储密码 ──────────────────────────────────
MINIO_ROOT_PASSWORD=MyStr0ng#MinIO#Pass

# ─── 4. JWT / Session 密钥（随机生成）──────────────────────
JWT_SECRET=（运行 openssl rand -hex 32 生成）
SESSION_SECRET=（运行 openssl rand -hex 32 生成）

# ─── 5. LLM 配置（客户本地大模型）──────────────────────────
GALAXY_AI_LLM_PROVIDER=openai_compatible
LLM_API_URL=http://192.168.1.200:8000/v1  # 客户 vLLM 地址
LLM_MODEL=qwen2.5-32b-instruct           # 客户部署的模型名
LLM_API_KEY=none                          # 内网通常不需要

# ─── 6. 工作流执行（本机 Docker）────────────────────────────
NEXTFLOW_USE_AWS=false

# ─── 7. Docker 镜像 ─────────────────────────────────────────
GALAXY_IMAGE=biomate-galaxy:latest
FRONTEND_IMAGE=biomate-frontend:latest

快速生成密钥

# 在服务器上执行，将输出复制到 .env
echo "JWT_SECRET=$(openssl rand -hex 32)"
echo "SESSION_SECRET=$(openssl rand -hex 32)"

步骤 7：运行安装脚本

root@server# cd /opt/biomate/biomate-src
root@server# bash install.sh

安装脚本自动完成：

  • 预检（Docker 版本 / 磁盘空间 / 内存 / 端口冲突）
  • 启动所有服务（postgres / redis / minio / galaxy / frontend）
  • 等待所有容器通过健康检查
  • 初始化数据库（执行 migration）
  • 创建管理员账号（交互输入邮箱和密码）
  • 验证所有 API 端点可访问
  • 打印访问地址
ℹ️
安装时间参考 首次安装约需 3~8 分钟（主要是数据库初始化和服务启动）。 安装日志保存在 /tmp/biomate-install.log。

6. 配置项详解

LLM 提供商选择

根据客户实际情况，选择以下方案之一：

场景配置项说明
客户自建本地模型
（vLLM / LMDeploy）		 
GALAXY_AI_LLM_PROVIDER=openai_compatible
LLM_API_URL=http://192.168.x.x:8000/v1
LLM_MODEL=qwen2.5-32b-instruct
LLM_API_KEY=none
 ✅ 推荐。纯内网，零数据出境
DeepSeek API
（内网可访问外网时）		 
GALAXY_AI_LLM_PROVIDER=deepseek
DEEPSEEK_API_KEY=sk-xxxx
 需要服务器能访问 api.deepseek.com
Ollama 本机运行
（无 GPU 或小型测试）		 
GALAXY_AI_LLM_PROVIDER=ollama
OLLAMA_BASE_URL=http://ollama:11434
 取消 docker-compose.onprem.yml 中 ollama 服务的注释

MinIO 存储

MinIO 是 BioMate 在本地部署中替代 AWS S3 的对象存储服务。工作流的输入文件、输出结果、中间文件均存储在 MinIO 中。

配置项默认值说明
MINIO_ROOT_USERbiomateMinIO 管理员用户名
MINIO_ROOT_PASSWORD（必须设置）MinIO 管理员密码，至少 8 位
S3_BUCKET_NAMEbiomate-data默认 bucket 名称

Nextflow 工作流执行

配置项默认值说明
NEXTFLOW_USE_AWSfalse单机模式设为 false，使用本机 Docker
NEXTFLOW_EXECUTORlocalsingle-node 不需要修改
NEXTFLOW_WORK_DIR/tmp/biomate-nf-workNextflow 工作目录（建议指向大磁盘）
⚠️
工作目录磁盘规划 NEXTFLOW_WORK_DIR 会存放所有工作流中间文件，RNA-seq 分析一次可能产生 50~200 GB 数据。 建议指向挂载的大容量磁盘，例如 /data/biomate-nf-work，并配置定期清理策略。

开机自启动

# 创建 systemd 服务，实现开机自动启动
cat > /etc/systemd/system/biomate.service << 'EOF'
[Unit]
Description=BioMate On-Premises Services
After=docker.service
Requires=docker.service

[Service]
Type=oneshot
RemainAfterExit=yes
WorkingDirectory=/opt/biomate/biomate-src
ExecStart=/usr/bin/docker compose -f docker-compose.onprem.yml up -d
ExecStop=/usr/bin/docker compose -f docker-compose.onprem.yml down
TimeoutStartSec=300

[Install]
WantedBy=multi-user.target
EOF

systemctl daemon-reload
systemctl enable biomate
systemctl start biomate  # 或重启服务器验证

数据备份

# 建议每天凌晨自动备份数据库和 MinIO 数据
cat > /opt/biomate/backup.sh << 'EOF'
#!/bin/bash
BACKUP_DIR="/data/biomate-backups/$(date +%Y%m%d)"
mkdir -p "$BACKUP_DIR"

# 备份 PostgreSQL
docker exec biomate-postgres pg_dump -U biomate biomate \
  | gzip > "$BACKUP_DIR/postgres.sql.gz"

# 备份 MinIO（用 mc mirror）
docker run --rm --network biomate-network \
  -v "$BACKUP_DIR:/backup" \
  minio/mc:latest \
  sh -c "mc alias set local http://biomate-minio:9000 \$MINIO_ROOT_USER \$MINIO_ROOT_PASSWORD \
    && mc mirror local/biomate-data /backup/minio-data"

# 保留最近 30 天
find /data/biomate-backups -maxdepth 1 -mtime +30 -exec rm -rf {} +
echo "备份完成: $BACKUP_DIR"
EOF

chmod +x /opt/biomate/backup.sh

# 添加到 crontab（每天凌晨 2 点执行）
(crontab -l 2>/dev/null; echo "0 2 * * * /opt/biomate/backup.sh >> /var/log/biomate-backup.log 2>&1") | crontab -

Nextflow 工作目录清理

# 工作流完成后的临时文件会在 NEXTFLOW_WORK_DIR 积累，建议定期清理
cat >> /opt/biomate/backup.sh << 'EOF'

# 清理 30 天前的 Nextflow 工作目录（保留结果，只删中间文件）
find /tmp/biomate-nf-work -maxdepth 2 -name ".command.*" -mtime +30 -delete
find /tmp/biomate-nf-work -maxdepth 2 -name "*.tmp" -mtime +30 -delete
EOF

9. 故障排查

服务无法启动

# 查看具体错误
bash scripts/biomate logs galaxy | tail -50

# 常见原因：
# 1. .env 必填项未设置
docker inspect biomate-galaxy | grep -A 5 "Error"

# 2. 端口被占用
ss -tlnp | grep -E ":5000|:8081|:9000"

# 3. 磁盘满
df -h

AI 无法回复（LLM 连接失败）

# 验证本地模型 API 可访问
curl http://LLM_API_URL/v1/models
# 应返回 {"object":"list","data":[...]}  如果 timeout 说明 LLM 服务地址不对

# 查看 Galaxy 日志中的 LLM 错误
bash scripts/biomate logs galaxy | grep -i "llm\|openai\|error" | tail -20

工作流提交失败

# 检查 docker.sock 挂载是否正常（Nextflow local 执行需要）
docker exec biomate-galaxy docker ps 2>&1

# 如果报 "permission denied on /var/run/docker.sock"
chmod 666 /var/run/docker.sock
# 或将 biomate-galaxy 容器的运行用户加入 docker 组

许可证验证失败

# 检查许可证配置
docker exec biomate-galaxy env | grep BIOMATE_LICENSE

# 重新触发验证
curl http://localhost:8081/api/license/status | python3 -m json.tool

MinIO 无法访问

# 检查 MinIO 状态
docker exec biomate-minio mc ready local

# 确认 bucket 存在
docker exec biomate-minio mc ls local/

恢复数据库备份

# 从备份恢复（注意：会覆盖当前数据库）
gunzip -c /data/biomate-backups/20260916/postgres.sql.gz \
  | docker exec -i biomate-postgres psql -U biomate biomate

10. 版本升级

在线升级（服务器可访问 BioMate 内部网络时）

# 使用升级脚本
bash scripts/onprem-update.sh --tag v2.6.0

离线升级（纯内网）

# 1. 在联网机器上下载新版本镜像
docker pull biomate-galaxy:v2.6.0
docker save biomate-galaxy:v2.6.0 | gzip > biomate-galaxy-v2.6.0.tar.gz

# 2. 传输到目标服务器
scp biomate-galaxy-v2.6.0.tar.gz user@目标服务器:/opt/

# 3. 在目标服务器加载并升级
docker load -i /opt/biomate-galaxy-v2.6.0.tar.gz
docker tag biomate-galaxy:v2.6.0 biomate-galaxy:latest

# 4. 重启 Galaxy 服务（停机 30 秒）
docker restart biomate-galaxy

# 5. 验证
bash scripts/biomate status
curl http://localhost:8081/api/version
✅
升级最佳实践 升级前执行一次手动备份：bash /opt/biomate/backup.sh
数据卷（PostgreSQL / MinIO）在升级过程中不会被删除，数据完全保留。
